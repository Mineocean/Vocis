# v0.4 本地 ASR 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将语音识别从云端 MiMo 改为本地 faster-whisper（默认 base 模型），实现离线、低延迟、多语言自动检测（中/英/日），中文源音频跳过翻译直接显示原文；DeepSeek 云端翻译保留。

**Architecture:** 音频采集与 VAD 切段、增量实时管道（方案 D）保持不变，仅替换 ASR 后端。faster-whisper 本地识别返回 `(text, language, probability)` 三元组；pipeline 增量 worker 根据检测语言决定是否跳过翻译（源语言==目标语言直接显示原文），低置信度时保守走翻译。

**Tech Stack:** Python 3.11、faster-whisper 1.2.1（本地模型 `models/faster-whisper-base`）、PySide6、pytest（mock 测试，不依赖真实模型）。

```python
import abc
import asyncio
from typing import Any


class ASREngine(abc.ABC):
    """ASR 引擎抽象基类。

    子类 transcribe 返回 `(text, language, probability) | None`：
      - text: 识别文本
      - language: 检测到的源语言代码（zh/en/ja；无法检测返回 None）
      - probability: 语言置信度 0.0~1.0
    """

    @abc.abstractmethod
    def transcribe(self, audio_pcm: bytes) -> tuple[str, str, float] | None:
        """识别 PCM 音频，返回三元组或 None。"""
        raise NotImplementedError

    async def transcribe_async(self, audio_pcm: bytes) -> tuple[str, str, float] | None:
        return await asyncio.to_thread(self.transcribe, audio_pcm)

    def set_language(self, language: str):
        """设置语言；'auto' 表示自动检测。"""

    def close(self):
        """释放资源。"""
```

```python
"""
Whisper ASR 引擎 —— faster-whisper 本地识别（多语言自动检测）。

返回 `(text, language, probability) | None` 三元组。
"""

import logging

import numpy as np

from .base import ASREngine
from .registry import register_asr
from .config import get_config

logger = logging.getLogger(__name__)


@register_asr("whisper")
class WhisperASR(ASREngine):
    def __init__(self, model_path: str | None = None, device: str | None = None):
        cfg = get_config().asr
        self._model_path = model_path or cfg.whisper_model_path
        self._device = device or cfg.whisper_device
        self._language: str | None = None
        self._model = None

    def set_language(self, language: str):
        self._language = None if language == "auto" else language

    def _load_model(self):
        if self._model is not None:
            return
        import faster_whisper
        compute = "int8"
        device = self._device
        if device in ("cuda", "auto"):
            try:
                import torch
                if torch.cuda.is_available():
                    compute = "float16"
                    device = "cuda"
                else:
                    device = "cpu"
            except ImportError:
                device = "cpu"
        logger.info("Loading Whisper model: %s (device=%s, compute=%s)", self._model_path, device, compute)
        self._model = faster_whisper.WhisperModel(self._model_path, device=device, compute_type=compute)

    def transcribe(self, audio_pcm: bytes) -> tuple[str, str, float] | None:
        if not audio_pcm:
            return None
        self._load_model()
        audio_np = np.frombuffer(audio_pcm, dtype=np.int16).astype(np.float32) / 32768.0
        max_samples = 16000 * 15
        if len(audio_np) > max_samples:
            audio_np = audio_np[-max_samples:]
        segments, info = self._model.transcribe(
            audio_np, language=self._language, beam_size=1, vad_filter=False
        )
        text = " ".join(seg.text.strip() for seg in segments).strip()
        if not text:
            return None
        return text, info.language, float(info.language_probability)

    def close(self):
        self._model = None
```

注意：`WhisperModel` 现在用 `whisper_model_path`（本地目录）而非模型 size。`whisper_model` 配置仍保留（可能用于模型 size 兜底，但主路径用 path）。

_LANG_MAP = {"zh": "中文", "en": "英文", "ja": "日文"}


# SubtitlePipeline 内新增方法（放在 _diff_text 之前）
    @staticmethod
    def _map_lang(lang: str | None) -> str:
        """检测语言代码 → 显示语言名（未知语言返回 '中文' 保守处理）。"""
        if not lang:
            return "中文"
        return _LANG_MAP.get(lang, "中文")

    def _should_skip_translate(self, lang: str | None, prob: float) -> bool:
        """源语言==目标语言时跳过翻译；低置信度保守走翻译。"""
        cfg = get_config()
        if not cfg.asr.skip_translate_when_same_lang:
            return False
        if prob < 0.5:
            return False
        mapped = self._map_lang(lang)
        return mapped == cfg.translator.target_language
```

def create_asr() -> ASREngine:
    """Create ASR engine from config; fall back to mimo if requested backend fails."""
    from ..config import get_config
    cfg = get_config()
    backend = cfg.asr.backend.lower()

    cls = _REGISTRY.get(backend)
    if cls is None:
        available = list(_REGISTRY.keys())
        logger.warning("Unknown ASR backend '%s', available: %s", backend, available)
        if "mock" in _REGISTRY:
            return _REGISTRY["mock"]()
        raise ValueError(f"Unknown ASR backend: {backend}. Available: {available}")

    try:
        return cls()
    except Exception as e:
        logger.warning("Failed to create ASR backend '%s': %s, falling back to mimo", backend, e)
        if backend != "mimo" and "mimo" in _REGISTRY:
            return _REGISTRY["mimo"]()
        raise
```
